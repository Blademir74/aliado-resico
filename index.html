💾 Datos

🗄️ SQL para Supabase

Ejecuta este SQL en el editor de Supabase para crear las tablas:

-- Historial de conversaciones
CREATE TABLE IF NOT EXISTS conversations (
  id TEXT PRIMARY KEY,
  text TEXT NOT NULL,
  sender TEXT DEFAULT 'Usuario',
  time TEXT,
  intent TEXT NOT NULL,
  confidence FLOAT,
  keywords TEXT[],
  explanation TEXT,
  response TEXT,
  source TEXT DEFAULT 'local',
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Documentos procesados (OCR)
CREATE TABLE IF NOT EXISTS documents (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  file_name TEXT,
  doc_type TEXT,
  extracted_data JSONB,
  confidence FLOAT,
  validation_status TEXT,
  needs_review BOOLEAN DEFAULT false,
  source TEXT DEFAULT 'unknown',
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Métricas fiscales
CREATE TABLE IF NOT EXISTS fiscal_metrics (
  id TEXT PRIMARY KEY DEFAULT 'primary',
  income_ytd NUMERIC DEFAULT 0,
  total_processed INT DEFAULT 0,
  by_category JSONB DEFAULT '{}',
  avg_confidence FLOAT DEFAULT 0,
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- Enable Realtime
ALTER PUBLICATION supabase_realtime
  ADD TABLE conversations;
ALTER PUBLICATION supabase_realtime
  ADD TABLE documents;
